BEGIN;

ALTER TABLE public.ocorrencias
  ADD COLUMN status VARCHAR(1) NOT NULL DEFAULT 'A',
  ADD CONSTRAINT ck_ocorrencias_status CHECK (status IN ('A', 'P', 'F'));

COMMENT ON COLUMN public.ocorrencias.status IS
  'A = Aberta; P = Em andamento; F = Finalizada. Transicoes, encerramento e permissoes ainda dependem de implementacao na API.';

-- Mantem colaborador obrigatorio nos desvios. Nao permite duplicar o mesmo
-- par colaborador/obra (PK original), mas permite obras diferentes.
-- Nao cria bloqueio novo para colaborador inativo: a API atual nao trata
-- essa rejeicao. A consulta abaixo fornece os dados para a futura validacao.
CREATE VIEW public.vw_colaboradores_resumo_obras
WITH (security_invoker = true) AS
SELECT
  c.id AS colaborador_id,
  c.empresa_id,
  c.status AS status_colaborador,
  c.status = 'Ativo' AS colaborador_ativo,
  count(co.obra_id)::integer AS quantidade_obras_vinculadas,
  count(co.obra_id) FILTER (WHERE o.status = 'Ativa')::integer
    AS quantidade_obras_ativas,
  count(co.obra_id) > 1 AS vinculado_a_multiplas_obras,
  count(co.obra_id) FILTER (WHERE o.status = 'Ativa') > 1
    AS vinculado_a_multiplas_obras_ativas
FROM public.colaboradores c
LEFT JOIN public.colaborador_obras co ON co.colaborador_id = c.id
LEFT JOIN public.obras o ON o.id = co.obra_id
GROUP BY c.id, c.empresa_id, c.status;

COMMENT ON VIEW public.vw_colaboradores_resumo_obras IS
  'Diagnostico de atividade e vinculos, sem bloquear multiplas obras. Nao constitui controle de autorizacao ou isolamento por organizacao.';
